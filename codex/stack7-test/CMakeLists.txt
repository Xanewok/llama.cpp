cmake_minimum_required(VERSION 3.16)
project(stack7_cpu_checks LANGUAGES C CXX)
set(LLAMA_SOURCE "${CMAKE_CURRENT_LIST_DIR}/../.." CACHE PATH "Source tree under test")
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
add_subdirectory("${LLAMA_SOURCE}" llama)
find_package(Threads REQUIRED)
foreach(fixture bw-np2-driver chain-cpu fold-cpu)
    add_executable(${fixture} "../${fixture}.cpp")
    target_compile_features(${fixture} PRIVATE cxx_std_17)
    target_include_directories(${fixture} PRIVATE "${LLAMA_SOURCE}/tools/server" "${LLAMA_SOURCE}/tools/mtmd" "${LLAMA_SOURCE}/vendor" "${LLAMA_SOURCE}")
    target_link_libraries(${fixture} PRIVATE server-context llama-common llama-common-base mtmd llama ggml Threads::Threads)
endforeach()
add_executable(outproj-cpu ../outproj-cpu.cpp)
target_compile_features(outproj-cpu PRIVATE cxx_std_17)
target_link_libraries(outproj-cpu PRIVATE llama ggml)
add_library(row0-mutant SHARED ../row0-mutant.cpp)
target_compile_features(row0-mutant PRIVATE cxx_std_17)
target_include_directories(row0-mutant PRIVATE "${LLAMA_SOURCE}/ggml/include")
target_link_libraries(row0-mutant PRIVATE ${CMAKE_DL_LIBS})
if(APPLE)
    target_link_options(row0-mutant PRIVATE -undefined dynamic_lookup)
endif()
set_target_properties(row0-mutant PROPERTIES PREFIX "" LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
