cmake_minimum_required(VERSION 3.16)
project(round_trace_test LANGUAGES CXX)
option(GGML_CUDA "Unused by the trace-only CPU test" OFF)
option(GGML_CCACHE "Use shared ccache" ON)
option(GGML_ROUND_TRACE_TSAN "Build only the ThreadSanitizer variant in a separate build directory" OFF)
if(GGML_CUDA)
    message(FATAL_ERROR "The trace test is CPU only")
endif()
if(GGML_CCACHE)
    find_program(CCACHE ccache REQUIRED)
    set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE}")
endif()
find_package(Threads REQUIRED)
set(variants patched old mutant)
if(GGML_ROUND_TRACE_TSAN)
    set(variants tsan)
endif()
foreach(variant ${variants})
    add_executable(trace-test-${variant} ../test-round-trace.cpp)
    target_compile_features(trace-test-${variant} PRIVATE cxx_std_17)
    target_include_directories(trace-test-${variant} PRIVATE ../../ggml/include)
    target_compile_options(trace-test-${variant} PRIVATE -Wall -Wextra -Werror)
    target_link_libraries(trace-test-${variant} PRIVATE Threads::Threads)
    if(variant STREQUAL "old" OR variant STREQUAL "mutant")
        target_compile_definitions(trace-test-${variant} PRIVATE GGML_ROUND_TRACE_TEST_SOURCE="${CMAKE_BINARY_DIR}/${variant}-round-trace.cpp")
    endif()
    if(NOT variant STREQUAL "old")
        target_compile_definitions(trace-test-${variant} PRIVATE GGML_ROUND_TRACE_TEST_HOOKS)
    endif()
    if(variant STREQUAL "tsan")
        target_compile_options(trace-test-${variant} PRIVATE -fsanitize=thread -fno-omit-frame-pointer -g)
        target_link_options(trace-test-${variant} PRIVATE -fsanitize=thread)
    endif()
endforeach()
