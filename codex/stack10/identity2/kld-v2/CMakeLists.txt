cmake_minimum_required(VERSION 3.16)
project(s10_kld LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
option(GGML_CCACHE "Use the shared compiler cache" ON)
option(GGML_CUDA "Build a CUDA backend" OFF)
if(GGML_CUDA)
    message(FATAL_ERROR "This tool links frozen libraries; it must not build a backend")
endif()
find_program(CCACHE ccache)
if(GGML_CCACHE AND CCACHE)
    set(CMAKE_CXX_COMPILER_LAUNCHER "${CCACHE}")
endif()
add_executable(s10-kld ../kld-window-v2.cpp)
target_include_directories(s10-kld PRIVATE "${ENGINE}/common" "${ENGINE}/include" "${ENGINE}/ggml/include" "${ENGINE}/src")
target_link_libraries(s10-kld PRIVATE "${FROZEN}/libllama-common.so" "${FROZEN}/libllama.so" "${FROZEN}/libggml.so" "${FROZEN}/libggml-base.so")
target_link_options(s10-kld PRIVATE "-Wl,-rpath-link,${FROZEN}")
set_target_properties(s10-kld PROPERTIES BUILD_RPATH "${FROZEN}")
